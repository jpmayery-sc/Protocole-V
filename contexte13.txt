def run_core(Nmax, use_coherence=False, seed=0):  
    rng = np.random.default_rng(seed)

    # Collecteurs  
    M_T0, M_T1, M_T2p = [], [], []  
    Q_counts = {-1:0, 0:0, +1:0}

    # classes n/p/e purement (Q,T)  
    M_n = []  # Q=0,T=0  
    M_p = []  # Q=+1,T=1  
    M_e = []  # Q=-1,T=1

    # Liste pour null-test (stockage de (M,T))  
    MT = []

    for m in range(5, Nmax+1):  
        for n in range(m+1, Nmax+1):  
            for k in range(n+1, Nmax+1):  
                if m+n <= k:  
                    continue  
                if use_coherence and not coherent_gcd(m,n,k):  
                    continue

                Q,T,A,M = QTM(m,n,k)  
                Q_counts[Q] += 1  
                MT.append((M,T,Q))

                if T==0: M_T0.append(M)  
                elif T==1: M_T1.append(M)  
                else: M_T2p.append(M)

                # n/p/e via (Q,T) uniquement  
                if Q==0 and T==0: M_n.append(M)  
                if Q==+1 and T==1: M_p.append(M)  
                if Q==-1 and T==1: M_e.append(M)

    def summary(arr):  
        arr = np.asarray(arr, dtype=float)  
        if len(arr)==0:  
            return dict(count=0, mean=np.nan, median=np.nan, ratio=np.nan)  
        return dict(count=len(arr),  
                    mean=float(arr.mean()),  
                    median=float(np.median(arr)),  
                    ratio=float(arr.mean()/np.median(arr)))

    sT0, sT1, sT2 = summary(M_T0), summary(M_T1), summary(M_T2p)  
    sn, sp, se = summary(M_n), summary(M_p), summary(M_e)

    # TEST 1: hiérarchie par T  
    ok_hierarchy_T = (sT0["mean"] > sT1["mean"] > sT2["mean"]) if all(np.isfinite([sT0["mean"],sT1["mean"],sT2["mean"]])) else False

    # TEST 4: symétrie Q  
    totQ = sum(Q_counts.values())  
    pQp = Q_counts[+1]/totQ if totQ else np.nan  
    pQm = Q_counts[-1]/totQ if totQ else np.nan  
    pQ0 = Q_counts[0]/totQ if totQ else np.nan  
    ok_Q_sym = (abs(pQp - pQm) < 0.02) if np.isfinite(pQp) else False  # tol 2%

    # Null test shuffle T : on mélange T dans MT et on refait la hiérarchie  
    Mvals = np.array([x[0] for x in MT])  
    Tvals = np.array([x[1] for x in MT])  
    rng.shuffle(Tvals)

    # recalcul moyennes conditionnelles après shuffle  
    def mean_cond(Tcond):  
        sel = Mvals[Tvals==Tcond]  
        return float(sel.mean()) if len(sel)>0 else np.nan

    mean0_s = mean_cond(0)  
    mean1_s = mean_cond(1)  
    mean2p_s = float(Mvals[Tvals>=2].mean()) if np.any(Tvals>=2) else np.nan  
    ok_shuffle_breaks = not (mean0_s > mean1_s > mean2p_s)  # on veut que la hiérarchie casse

    return dict(  
        Nmax=Nmax,  
        coherence=use_coherence,  
        # T-classes  
        T0_count=sT0["count"], T1_count=sT1["count"], T2p_count=sT2["count"],  
        T0_mean=sT0["mean"],  T1_mean=sT1["mean"],  T2p_mean=sT2["mean"],  
        T0_ratio=sT0["ratio"],  
        ok_hierarchy_T=ok_hierarchy_T,  
        # Q symmetry  
        pQplus=pQp, pQminus=pQm, pQzero=pQ0,  
        ok_Q_sym=ok_Q_sym,  
        # n/p/e via (Q,T)  
        n_count=sn["count"], p_count=sp["count"], e_count=se["count"],  
        n_mean=sn["mean"], p_mean=sp["mean"], e_mean=se["mean"],  
        # null test  
        shuffle_T0_mean=mean0_s, shuffle_T1_mean=mean1_s, shuffle_T2p_mean=mean2p_s,  
        ok_shuffle_breaks=ok_shuffle_breaks

def score_from_labels(Mvals, labels):  
    # labels: 0,1,2 (2 means >=2)  
    mu0 = Mvals[labels==0].mean()  
    mu1 = Mvals[labels==1].mean()  
    mu2 = Mvals[labels==2].mean()  
    return (mu0 - mu1) + (mu1 - mu2)  # = mu0 - mu2

def pvalue_shuffle(Nmax, use_coherence=False, nshuffle=200, seed=0):  
    rng = np.random.default_rng(seed)  
    MT = []

    for m in range(5, Nmax+1):  
        for n in range(m+1, Nmax+1):  
            for k in range(n+1, Nmax+1):  
                if m+n <= k:  
                    continue  
                if use_coherence and not coherent_gcd(m,n,k):  
                    continue  
                Q,T,A,M = QTM(m,n,k)  
                lab = 0 if T==0 else (1 if T==1 else 2)  
                MT.append((M, lab))

    Mvals = np.array([x[0] for x in MT], dtype=float)  
    labs  = np.array([x[1] for x in MT], dtype=int)

    S_real = score_from_labels(Mvals, labs)

    S_shuf = []  
    for _ in range(nshuffle):  
        labs2 = labs.copy()  
        rng.shuffle(labs2)  
        S_shuf.append(score_from_labels(Mvals, labs2))

    S_shuf = np.array(S_shuf)  
    p = np.mean(S_shuf >= S_real)  
    return S_real, S_shuf.mean(), S_shuf.std(), p

def Q_distribution_triplets(Nmax, use_coherence=False):  
    counts = {-1:0, 0:0, +1:0}  
    for m in range(5, Nmax+1):  
        for n in range(m+1, Nmax+1):  
            for k in range(n+1, Nmax+1):  
                if m+n <= k:   
                    continue  
                if use_coherence and not coherent_gcd(m,n,k):  
                    continue  
                Q,T,A,M = QTM(m,n,k)  
                counts[Q]+=1  
    tot = sum(counts.values())  
    return {q: counts[q]/tot for q in counts}

def Q_distribution_increments(Nmax, dmax=50, seed=0):  
    # Tirage uniforme sur (m,d1,d2) sous contrainte triangle : m+(m+d1) > (m+d1+d2)  
    # => m > d2  
    rng = np.random.default_rng(seed)  
    counts = {-1:0, 0:0, +1:0}  
    ntrials = 200000  
    for _ in range(ntrials):  
        m  = rng.integers(5, Nmax+1)  
        d1 = rng.integers(1, dmax+1)  
        d2 = rng.integers(1, dmax+1)  
        if m <= d2:  
            continue  
        Q = 0 if d2==d1 else (1 if d2>d1 else -1)  
        counts[Q]+=1  
    tot = sum(counts.values())  
    return {q: counts[q]/tot for q in counts}
