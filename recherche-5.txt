import math  
import statistics as stats

# ----------------------------  
# Paramètres  
# ----------------------------  
N = 60  # fenêtre: m < n < k <= N

PETITS = {5, 7, 11, 13}

# seuils de classe (canon)  
T_NEUTRON_MAX = 1  
ASPECT_NEUTRON_MAX = 2

ASPECT_PROTON_MAX = 3

ASPECT_ELECTRON_MIN = 4

LAMBDA_PHYS = 1836.15267389

# ----------------------------  
# Outils  
# ----------------------------  
def gcd(a, b):  
    return math.gcd(a, b)

def coherent_gcd(m, n, k):  
    gmn = gcd(m, n)  
    gnk = gcd(n, k)  
    gmk = gcd(m, k)  
    return (gmn == gnk == gmk)

def heron_area(m, n, k):  
    s = 0.5 * (m + n + k)  
    val = s * (s - m) * (s - n) * (s - k)  
    if val <= 0:  
        return 0.0  
    return math.sqrt(val)

def Q_T(m, n, k):  
    d1 = n - m  
    d2 = k - n  
    diff = d2 - d1  
    Q = 0  
    if diff > 0:  
        Q = +1  
    elif diff < 0:  
        Q = -1  
    T = abs(diff)  
    return Q, T

def mass_geo(m, n, k):  
    Q, T = Q_T(m, n, k)  
    A = heron_area(m, n, k)  
    return A / (1.0 + T*T)

def aspect(m, n, k):  
    return k / m

def count_petits(m, n, k):  
    return int(m in PETITS) + int(n in PETITS) + int(k in PETITS)

def has_any_petit(m, n, k):  
    return (m in PETITS) or (n in PETITS) or (k in PETITS)

# ----------------------------  
# Classes  
# ----------------------------  
def is_neutron_like(m, n, k):  
    if not coherent_gcd(m, n, k):  
        return False  
    if has_any_petit(m, n, k):  
        return False  
    Q, T = Q_T(m, n, k)  
    if T > T_NEUTRON_MAX:  
        return False  
    if aspect(m, n, k) > ASPECT_NEUTRON_MAX:  
        return False  
    return True

def is_proton_like(m, n, k):  
    if not coherent_gcd(m, n, k):  
        return False  
    if count_petits(m, n, k) != 1:  
        return False  
    Q, T = Q_T(m, n, k)  
    if Q != +1:  
        return False  
    if aspect(m, n, k) > ASPECT_PROTON_MAX:  
        return False  
    return True

def is_electron_like(m, n, k):  
    if not coherent_gcd(m, n, k):  
        return False  
    if count_petits(m, n, k) != 1:  
        return False  
    Q, T = Q_T(m, n, k)  
    if Q != -1:  
        return False  
    if aspect(m, n, k) < ASPECT_ELECTRON_MIN:  
        return False  
    return True

# ----------------------------  
# Scan  
# ----------------------------  
Cn, Cp, Ce = [], [], []

for m in range(1, N+1):  
    for n in range(m+1, N+1):  
        for k in range(n+1, N+1):  
            # triangle inequality  
            if m + n <= k:  
                continue  
            if is_neutron_like(m, n, k):  
                Cn.append((m, n, k))  
            if is_proton_like(m, n, k):  
                Cp.append((m, n, k))  
            if is_electron_like(m, n, k):  
                Ce.append((m, n, k))

def masses_of(triplets):  
    return [mass_geo(m, n, k) for (m, n, k) in triplets]

Mn_list = masses_of(Cn)  
Mp_list = masses_of(Cp)  
Me_list = masses_of(Ce)

def safe_mean(x):  
    return sum(x)/len(x) if len(x) else float("nan")

def safe_median(x):  
    return stats.median(x) if len(x) else float("nan")

Mn = safe_mean(Mn_list)  
Mp = safe_mean(Mp_list)  
Me = safe_mean(Me_list)

median_Mn = safe_median(Mn_list)  
Cint_n = Mn/median_Mn if median_Mn and not math.isnan(Mn) else float("nan")

lambda_geo = Mp/Me if Me and not math.isnan(Mp) and not math.isnan(Me) else float("nan")  
lambda_pred = 1728.0 * Cint_n if not math.isnan(Cint_n) else float("nan")  
eps = (LAMBDA_PHYS - lambda_pred) / LAMBDA_PHYS if not math.isnan(lambda_pred) else float("nan")

# ----------------------------  
# Résultats  
# ----------------------------  
print("=== Fenêtre ===")  
print(f"N = {N} (m<n<k<=N)")

print("\n=== Cardinaux ===")  
print(f"|Cn| = {len(Cn)}")  
print(f"|Cp| = {len(Cp)}")  
print(f"|Ce| = {len(Ce)}")

print("\n=== Masses géométriques (moyennes) ===")  
print(f"Mn = {Mn}")  
print(f"Mp = {Mp}")  
print(f"Me = {Me}")

print("\n=== Intrication neutrons (statistique) ===")  
print(f"median(Mn) = {median_Mn}")  
print(f"Cint^(n) = mean/median = {Cint_n}")

print("\n=== Ratios ===")  
print(f"lambda_geo = Mp/Me = {lambda_geo}")  
print(f"lambda_pred = 1728*Cint^(n) = {lambda_pred}")  
print(f"lambda_phys = {LAMBDA_PHYS}")  
print(f"erreur relative eps = (phys - pred)/phys = {eps}")

# (Optionnel) quelques exemples pour contrôle  
print("\nExemples (5) Cn:", Cn[:5])  
print("Exemples (5) Cp:", Cp[:5])  
print("Exemples (5) Ce:", Ce[:5])
