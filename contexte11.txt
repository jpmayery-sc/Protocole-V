import numpy as np

G = 6.67430e-11  
c = 299792458.0  
M_sun = 1.98847e30  
l_pl = 1.616255e-35

def r_s(M):  
    return 2*G*M/c**2

def echo_delay(M, eps):  
    rs = r_s(M)  
    return 2*(rs/c)*np.log(rs/eps)

masses = [  
    ("10 Msun", 10*M_sun),  
    ("30 Msun", 30*M_sun),  
    ("2.83 Msun", 2.83*M_sun),  
    ("SgrA* 4.3e6 Msun", 4.3e6*M_sun),  
]

eps_list = [  
    ("eps = l_Pl", l_pl),  
    ("eps = 1e-20 m", 1e-20),  
    ("eps = rs/1e11", None),  # filled per mass  
]

for name, M in masses:  
    rs = r_s(M)  
    print("\nMass:", name, "rs =", rs, "m")  
    for ename, eps in eps_list:  
        if eps is None:  
            eps_use = rs/1e11  
            en = f"{ename} -> {eps_use:.3e} m"  
        else:  
            eps_use = eps  
            en = ename  
        dt = echo_delay(M, eps_use)  
        print(" ", en, "Δt_echo =", dt, "s")

import numpy as np  
import matplotlib.pyplot as plt

# -----------------------  
# Echo model (toy)  
# -----------------------  
def ringdown(t, f0=250, tau=0.02, A=1.0, phi0=0.0):  
    # exponentially damped sinusoid  
    return A*np.exp(-t/tau)*np.sin(2*np.pi*f0*t + phi0)

def add_echoes(t, base, dt_echo=0.02, n_echo=6, r=0.35, stretch=1.0):  
    """  
    Add echoes delayed by dt_echo with amplitude ratio r^k.  
    stretch>1 stretches the damping time slightly for later echoes (optional).  
    """  
    h = base.copy()  
    for k in range(1, n_echo+1):  
        shift = k*dt_echo  
        # shifted signal  
        tk = t - shift  
        mk = tk >= 0  
        hk = np.zeros_like(t)  
        # reuse ringdown form by sampling base at shifted time  
        hk[mk] = np.interp(tk[mk], t, base)  
        h += (r**k)*hk  
    return h

def corr_score(h, dt, D):  
    """  
    Simple normalized correlation between h(t) and h(t+D).  
    """  
    shift = int(round(D/dt))  
    if shift <= 0 or shift >= len(h)-10:  
        return np.nan  
    x = h[:-shift]  
    y = h[shift:]  
    # normalize  
    x = x - x.mean()  
    y = y - y.mean()  
    den = np.sqrt(np.sum(x*x)*np.sum(y*y))  
    if den == 0:  
        return np.nan

import numpy as np  
import matplotlib.pyplot as plt

# -----------------------  
# Toy echo signal  
# -----------------------  
def ringdown(t, f0=250, tau=0.02, A=1.0, phi0=0.0):  
    return A*np.exp(-t/tau)*np.sin(2*np.pi*f0*t + phi0)

def add_echoes(t, base, dt_echo=0.02, n_echo=6, r=0.35):  
    h = base.copy()  
    for k in range(1, n_echo+1):  
        shift = k*dt_echo  
        tk = t - shift  
        mk = tk >= 0  
        hk = np.zeros_like(t)  
        hk[mk] = np.interp(tk[mk], t, base)  
        h += (r**k)*hk  
    return h

# simple bandpass via FFT masking (no scipy)  
def bandpass_fft(x, fs, f_lo, f_hi):  
    X = np.fft.rfft(x)  
    freqs = np.fft.rfftfreq(len(x), d=1/fs)  
    mask = (freqs >= f_lo) & (freqs <= f_hi)  
    X *= mask  
    return np.fft.irfft(X, n=len(x))

def corr_norm(x, y):  
    x = x - x.mean()  
    y = y - y.mean()  
    den = np.sqrt(np.sum(x*x)*np.sum(y*y))  
    if den == 0:  
        return 0.0  
    return float(np.sum(x*y)/den)

def multi_echo_score(h, fs, D, K=6, decay=0.35):  
    """  
    Score(D) = sum_{k=1..K} decay^k * corr(h(t), h(t+kD))  
    """  
    dt = 1/fs  
    score = 0.0  
    for k in range(1, K+1):  
        shift = int(round((k*D)/dt))  
        if shift <= 0 or shift >= len(h)-10:  
            break  
        x = h[:-shift]  
        y = h[shift:]  
        score += (decay**k)*corr_norm(x, y)  
    return score

    X = np.fft.rfft(x)

def hann_taper(x, frac=0.05):  
    n = len(x)  
    m = int(frac*n)  
    if m < 2: return x  
    w = np.ones(n)  
    win = 0.5*(1-np.cos(np.pi*np.arange(m)/m))  
    w[:m] = win  
    w[-m:] = win[::-1]  
    return x*w

def simple_whiten(x, fs, f_lo=20, f_hi=1024, floor=1e-12):  
    """  
    Very simplified whitening:  
      whiten in frequency domain by dividing by sqrt(PSD) estimated from |X|^2 itself.  
    This is not as good as using off-source PSD, but helps for a quick pass.  
    """  
    X = np.fft.rfft(x)  
    freqs = np.fft.rfftfreq(len(x), d=1/fs)  
    psd = (np.abs(X)**2)  
    psd = np.maximum(psd, floor*np.max(psd))  
    W = X/np.sqrt(psd)  
    # band-limit  
    mask = (freqs >= f_lo) & (freqs <= f_hi)  
    W *= mask  
    return np.fft.irfft(W, n=len(x))

def corr_norm(x, y):  
    x = x - x.mean()  
    y = y - y.mean()  
    den = np.sqrt(np.sum(x*x)*np.sum(y*y))  
    if den == 0:  
        return 0.0  
    return float(np.sum(x*y)/den)

def bandpass_fft(x, fs, f_lo, f_hi):
X = np.fft.rfft(x)
freqs = np.fft.rfftfreq(len(x), d=1/fs)
mask = (freqs >= f_lo) & (freqs <= f_hi)
X *= mask
return np.fft.irfft(X, n=len(x))

def hann_taper(x, frac=0.05):
n = len(x)
m = int(fracn)
if m < 2: return x

def simple_whiten(x, fs, f_lo=20, f_hi=1024, floor=1e-12):
"""
Very simplified whitening:
whiten in frequency domain by dividing by sqrt(PSD) estimated from |X|^2 itself.
This is not as good as using off-source PSD, but helps for a quick pass.
"""
X = np.fft.rfft(x)
freqs = np.fft.rfftfreq(len(x), d=1/fs)
psd = (np.abs(X)*2)
psd = np.maximum(psd, floornp.max(psd))
W = X/np.sqrt(psd)
# band-limit
mask = (freqs >= f_lo) & (freqs <= f_hi)
W *= mask
return np.fft.irfft(W, n=len(x))

def corr_norm(x, y):
x = x - x.mean()
y = y - y.mean()
den = np.sqrt(np.sum(xx)np.sum(yy))
if den == 0:
return 0.0
return float(np.sum(xy)/den)

def multi_echo_score(h, fs, D, K=6, decay=0.35):
dt = 1/fs
score = 0.0
for k in range(1, K+1):
shift = int(round((k*D)/dt))
if shift <= 0 or shift >= len(h)-10:
break
x = h[:-shift]
y = h[shift:]
score += (decay**k)*corr_norm(x, y)
return score

# Files expected in ./sample_data/ :

#  2) crop around merger GPS (approx)

#  4) post-merger window

#  6) Monte Carlo null by circular roll

import h5py

# constants + prediction  
# -----------------------  
G = 6.67430e-11  
c = 299792458.0  
M_sun = 1.98847e30  
l_pl = 1.616255e-35

# choose epsilon prescription  
EPS_MODE = "planck"       # "planck" or "rs_over_1e11"  
Lambda_geom = 1e11

def r_s(M): return 2*G*M/c**2  
def echo_delay(M, eps):  
    rs = r_s(M)  
    return 2*(rs/c)*np.log(rs/eps)

# GW150914 final mass ~ 62 Msun (order-of-magnitude; change if you want)  
M_final = 62*M_sun  
rs = r_s(M_final)  
eps = l_pl if EPS_MODE=="planck" else rs/Lambda_geom  
dt_echo_pred = echo_delay(M_final, eps)  
print("Predicted echo delay Δt_echo =", dt_echo_pred, "s   (EPS_MODE =", EPS_MODE, ")")

# Approx GPS merger time (GW150914)  
gps_event = 1126259462.4

# -----------------------  
# DSP helpers  
# -----------------------  
def hann_taper(x, frac=0.05):  
    n = len(x)  
    m = int(frac*n)  
    if m < 2:  
        return x  
    w = np.ones(n)  
    win = 0.5*(1-np.cos(np.pi*np.arange(m)/m))  
    w[:m] = win  
    w[-m:] = win[::-1]  
    return x*w

def bandpass_fft(x, fs, f_lo, f_hi):  
    X = np.fft.rfft(x)  
    freqs = np.fft.rfftfreq(len(x), d=1/fs)  
    mask = (freqs >= f_lo) & (freqs <= f_hi)  
    X *= mask  
    return np.fft.irfft(X, n=len(x))

def simple_whiten(x, fs, f_lo=20, f_hi=1024, floor=1e-12):  
    """  
    Very simplified whitening using the segment itself.  
    For a better PSD, use off-source data; this is a quick test only.  
    """  
    X = np.fft.rfft(x)  
    freqs = np.fft.rfftfreq(len(x), d=1/fs)  
    psd = np.abs(X)**2  
    psd = np.maximum(psd, floor*np.max(psd))  
    W = X/np.sqrt(psd)  
    mask = (freqs >= f_lo) & (freqs <= f_hi)  
    W *= mask  
    return np.fft.irfft(W, n=len(x))

def corr_norm(x, y):  
    x = x - x.mean()  
    y = y - y.mean()  
    den = np.sqrt(np.sum(x*x)*np.sum(y*y))  
    if den == 0:  
        return 0.0  
    return float(np.sum(x*y)/den)

def multi_echo_score(h, fs, D, K=6, decay=0.35):  
    dt = 1/fs  
    score = 0.0  
    for k in range(1, K+1):  
        shift = int(round((k*D)/dt))  
        if shift <= 0 or shift >= len(h)-10:  
            break  
        score += (decay**k)*corr_norm(h[:-shift], h[shift:])  
    return score

# -----------------------  
# Load local HDF5  
# -----------------------  
def load_gw150914_hdf5(path):  
    with h5py.File(path, "r") as f:  
        # common GWOSC layout  
        strain = f["strain/Strain"][()].astype(float)  
        meta = dict(f["meta"].attrs)  
        fs = float(meta["SampleRate"])  
        gps_start = float(meta["GPSstart"])  
    t = gps_start + np.arange(len(strain))/fs  
    return t, strain, fs, gps_start

path_l1 = "sample_data/GW150914_L1_32s.hdf5"

tH, hH, fsH, gpsH = load_gw150914_hdf5(path_h1)

# align time window intersection

t1 = min(tH[-1], tL[-1])  
maskH = (tH >= t0) & (tH <= t1)  
maskL = (tL >= t0) & (tL <= t1)

t = tH[maskH]

hL = hann_taper(hL, 0.05)

print("Post-merger window:", tw[0], "to", tw[-1], "s ; samples:", len(hw))

# -----------------------  
# Scan delays

D_grid = np.linspace(0.002, 0.12, 700)  
S = np.array([multi_echo_score(hw, fs, D, K=6, decay=0.35) for D in D_grid])

best_i = np.argmax(S)  
D_best = D_grid[best_i]

print("  null mean±std:", S_null.mean(), S_null.std())  
print("  p-value at predicted delay:", p_val)

plt.figure(figsize=(12,4))

def inspect_h5(path, max_lines=200):  
    out = []  
    def visit(name, obj):  
        if isinstance(obj, h5py.Dataset):  
            out.append(("DATASET", name, obj.shape, str(obj.dtype)))  
        elif isinstance(obj, h5py.Group):  
            out.append(("GROUP", name, None, None))

    print("\n=== CONTENTS (first entries) ===")  
    for row in out[:max_lines]:  
        print(*row)  
    if len(out) > max_lines:

import h5py, numpy as np

def find_first_existing(f, candidates):  
    for c in candidates:  
        if c in f:  
            return c  
    return None

def load_strain_any(path):  
    with h5py.File(path, "r") as f:  
        # 1) find strain dataset  
        strain_key = find_first_existing(f, [  
            "strain/Strain", "strain/strain", "Strain", "strain",  
            "h", "data/strain", "data/Strain"  
        ])  
        if strain_key is None:  
            # fallback: pick the longest 1D dataset  
            best = None  
            best_len = 0  
            def visit(name, obj):  
                nonlocal best, best_len  
                if isinstance(obj, h5py.Dataset) and len(obj.shape)==1:  
                    if obj.shape[0] > best_len:  
                        best_len = obj.shape[0]  
                        best = name  
            f.visititems(visit)  
            strain_key = best

        if strain_key is None:  
            raise RuntimeError("Could not find a 1D strain dataset in this file.")

        h = f[strain_key][()].astype(float)

        # 2) sampling rate / dt  
        fs = None  
        gps_start = None

        # common attribute locations  
        attr_sources = [f.attrs]  
        if "meta" in f:  
            attr_sources.append(f["meta"].attrs)  
        if "strain" in f:  
            attr_sources.append(f["strain"].attrs)

        # search for known keys  
        for attrs in attr_sources:  
            for key in ["SampleRate", "sample_rate", "fs", "Fs", "SamplingRate", "Sampling_rate"]:  
                if key in attrs:  
                    fs = float(attrs[key])  
            for key in ["GPSstart", "gps_start", "start_time", "StartTimeGPS", "t0", "time_0"]:  
                if key in attrs:  
                    gps_start = float(attrs[key])

        # 3) if time vector exists, use it  
        time_key = find_first_existing(f, ["strain/Time", "time", "Time", "strain/time", "t"])  
        if time_key is not None:  
            t = f[time_key][()].astype(float)  
            if fs is None:  
                dt = np.median(np.diff(t))  
                fs = 1.0/dt  
            if gps_start is None:  
                gps_start = float(t[0])  
            # return consistent time axis  
            return t, h, fs, gps_start, strain_key, time_key

        # 4) otherwise, must have fs and gps_start  
        if fs is None:  
            raise RuntimeError(f"Sampling rate not found in attrs. Need fs. strain_key={strain_key}")  
        if gps_start is None:  
            # if missing, we can still build a relative time axis  
            gps_start = 0.0

        t = gps_start + np.arange(len(h))/fs  
        return t, h, fs, gps_start, strain_key, None

def hann_taper(x, frac=0.05):  
    n = len(x)  
    m = int(frac*n)  
    if m < 2:  
        return x  
    w = np.ones(n)  
    win = 0.5*(1-np.cos(np.pi*np.arange(m)/m))  
    w[:m] = win  
    w[-m:] = win[::-1]  
    return x*w

def bandpass_fft(x, fs, f_lo, f_hi):  
    X = np.fft.rfft(x)  
    freqs = np.fft.rfftfreq(len(x), d=1/fs)  
    mask = (freqs >= f_lo) & (freqs <= f_hi)  
    X *= mask  
    return np.fft.irfft(X, n=len(x))

def simple_whiten(x, fs, f_lo=20, f_hi=1024, floor=1e-12):  
    X = np.fft.rfft(x)  
    freqs = np.fft.rfftfreq(len(x), d=1/fs)  
    psd = np.abs(X)**2  
    psd = np.maximum(psd, floor*np.max(psd))  
    W = X/np.sqrt(psd)  
    mask = (freqs >= f_lo) & (freqs <= f_hi)  
    W *= mask  
    return np.fft.irfft(W, n=len(x))

def corr_norm(x, y):  
    x = x - x.mean()  
    y = y - y.mean()  
    den = np.sqrt(np.sum(x*x)*np.sum(y*y))  
    if den == 0:  
        return 0.0  
    return float(np.sum(x*y)/den)

def multi_echo_score(h, fs, D, K=6, decay=0.35):  
    dt = 1/fs  
    score = 0.0  
    for k in range(1, K+1):  
        shift = int(round((k*D)/dt))  
        if shift <= 0 or shift >= len(h)-10:  
            break  
        score += (decay**k)*corr_norm(h[:-shift], h[shift:])  
    return score

1) Traduction en paramètre testable : réflectivité (\mathcal{R})

    (\mathcal{R}\approx 0) : absorbeur (trou noir GR-like) → pas d’échos
    (\mathcal{R}>0) : miroir partiel → échos possibles

Ton test GW150914 donne une contrainte qualitative : (\mathcal{R}) doit être petit.
2) Test plus adapté à un absorbeur : QNM (ringdown) plutôt que échos

3) Un test encore plus direct : Love numbers

Donc si ton pont est absorbant et “trou noir vrai”, tu prédis Love numbers ~0.

# GW150914 — Upper limit on echo reflectivity r (toy calibration)

# Idea:

#  - Measure S_obs = multi-echo score at the predicted delay D_pred  
#  - Build many toy signals = (ringdown + echoes with reflectivity r) + noise  
#    passed through the SAME filter+score pipeline  
#  - For each r, estimate P(S_toy <= S_obs). If this is ~0.95, r is excluded at 95%.  
#  
# This gives an "upper limit" on r in the context of this toy echo model.  
#  
# Requirements:  
#  - Your local files exist:  
#     sample_data/GW150914_H1_32s.hdf5

import numpy as np  
import h5py  
import matplotlib.pyplot as plt

# -----------------------  
# Settings (match previous pipeline)  
# -----------------------  
EPS_MODE = "planck"      # just for printing; D_pred is computed from M_final and eps choice below  
Lambda_geom = 1e11  
G = 6.67430e-11  
c = 299792458.0  
M_sun = 1.98847e30  
l_pl = 1.616255e-35

gps_event = 1126259462.4  
post_start = 0.02  
post_end   = 0.30

# echo score params  
K_echo = 6  
decay_for_score = None   # we will use r itself as decay in the toy echoes  
# but in the score we still weight by decay^k; we'll set decay= r for consistency.

# bandpass/whiten params  
BP_LO, BP_HI = 30, 600  
W_LO, W_HI = 20, 1024  
TAPER_FRAC = 0.05

# Mass assumption for predicted delay  
M_final = 62*M_sun  
def r_s(M): return 2*G*M/c**2  
def echo_delay(M, eps):  
    rs = r_s(M)  
    return 2*(rs/c)*np.log(rs/eps)

rs = r_s(M_final)  
eps = l_pl  # change to rs/Lambda_geom if you want the other prescription  
D_pred = echo_delay(M_final, eps)  
print("D_pred =", D_pred, "s (EPS_MODE assumed =", EPS_MODE, ")")

# -----------------------  
# DSP helpers  
# -----------------------  
def hann_taper(x, frac=0.05):  
    n = len(x)  
    m = int(frac*n)  
    if m < 2:  
        return x  
    w = np.ones(n)  
    win = 0.5*(1-np.cos(np.pi*np.arange(m)/m))  
    w[:m] = win  
    w[-m:] = win[::-1]  
    return x*w

def bandpass_fft(x, fs, f_lo, f_hi):  
    X = np.fft.rfft(x)  
    freqs = np.fft.rfftfreq(len(x), d=1/fs)  
    mask = (freqs >= f_lo) & (freqs <= f_hi)  
    X *= mask  
    return np.fft.irfft(X, n=len(x))

def simple_whiten(x, fs, f_lo=20, f_hi=1024, floor=1e-12):  
    X = np.fft.rfft(x)  
    freqs = np.fft.rfftfreq(len(x), d=1/fs)  
    psd = np.abs(X)**2  
    psd = np.maximum(psd, floor*np.max(psd))  
    W = X/np.sqrt(psd)  
    mask = (freqs >= f_lo) & (freqs <= f_hi)  
    W *= mask  
    return np.fft.irfft(W, n=len(x))

def corr_norm(x, y):  
    x = x - x.mean()  
    y = y - y.mean()  
    den = np.sqrt(np.sum(x*x)*np.sum(y*y))  
    if den == 0:  
        return 0.0  
    return float(np.sum(x*y)/den)

def multi_echo_score(h, fs, D, K=6, decay=0.35):  
    dt = 1/fs  
    score = 0.0  
    for k in range(1, K+1):  
        shift = int(round((k*D)/dt))  
        if shift <= 0 or shift >= len(h)-10:  
            break  
        score += (decay**k)*corr_norm(h[:-shift], h[shift:])  
    return score

# Load and process GW150914 strain (H1 and L1)  
# -----------------------  
def load_gwosc_32s(path):  
    with h5py.File(path, "r") as f:  
        h = f["strain/Strain"][()].astype(float)  
        gps_start = float(f["meta/GPSstart"][()])  
        duration = float(f["meta/Duration"][()])  
        fs = len(h)/duration  
    t = gps_start + np.arange(len(h))/fs  
    return t, h, fs

def process_strain(t, h, fs):  
    h = hann_taper(h, TAPER_FRAC)  
    h = simple_whiten(h, fs, W_LO, W_HI)  
    h = bandpass_fft(h, fs, BP_LO, BP_HI)  
    return h

print("Observed S_pred (net) =", S_obs)

# -----------------------  
# Build a toy ringdown template (simple damped sinusoid) aligned at window start  
# We'll inject echoes into this template and add noise from the real data (bootstrap).  
# -----------------------  
def ringdown(t, f0=250, tau=0.02, A=1.0, phi0=0.0):  
    return A*np.exp(-t/tau)*np.sin(2*np.pi*f0*t + phi0)

def add_echoes(t, base, D, n_echo=6, r=0.3):  
    h = base.copy()  
    for k in range(1, n_echo+1):  
        shift = k*D  
        tk = t - shift  
        mk = tk >= 0  
        hk = np.zeros_like(t)  
        hk[mk] = np.interp(tk[mk], t, base)  
        h += (r**k)*hk  
    return h

# time axis in window starting at 0  
twin = tw - tw[0]

# crude ringdown params (not a fit): just a reasonable band  
base = ringdown(twin, f0=250, tau=0.02, A=1.0, phi0=0.0)  
# normalize base to match the scale of processed data  
scale = np.std(h1 + l1)  
base = base / (np.std(base) + 1e-12) * scale

# Use real processed data as noise proxy by random circular shifting (bootstrap)  
rng = np.random.default_rng(0)

def bootstrap_noise(x):  
    return np.roll(x, rng.integers(0, len(x)))

# -----------------------  
# Scan r values and compute exclusion  
# -----------------------  
r_grid = np.linspace(0.0, 0.9, 19)   # coarse grid first  
nmc = 400  # increase to 2000 for final limits

results = []  
for r in r_grid:  
    S_sim = np.empty(nmc)  
    # build injected signal once (same for H1 and L1 for simplicity)  
    inj = add_echoes(twin, base, D_pred, n_echo=6, r=r)

    for i in range(nmc):  
        # bootstrap noise from real data  
        n1 = bootstrap_noise(h1)  
        n2 = bootstrap_noise(l1)

        # inject same echo signal into both channels (very conservative/simple)  
        h1_sim = n1 + inj  
        l1_sim = n2 + inj

        # compute net score using same score decay (use r as "echo physics", but keep score decay fixed at 0.35)  
        S_sim[i] = (multi_echo_score(h1_sim, fs, D_pred, K=K_echo, decay=0.35) +  
                    multi_echo_score(l1_sim, fs, D_pred, K=K_echo, decay=0.35))

    # if echoes exist with reflectivity r, we'd expect S_sim typically > S_obs.  
    # Compute exclusion confidence: P(S_sim <= S_obs)  
    p_le = (np.sum(S_sim <= S_obs) + 1)/(nmc + 1)  
    results.append((r, p_le, np.mean(S_sim), np.std(S_sim)))

print("\n=== Exclusion scan (higher p_le means r is more strongly excluded) ===")  
for r, p_le, mu, sd in results:

# Find an approximate r_95 where p_le ~ 0.95  
cand = [x for x in results if x[1] >= 0.95]  
if cand:

else:
