sigma_z = np.array([[1, 0],

                    [0,-1]], dtype=complex)

I2 = np.eye(2, dtype=complex)

#    H = c p σ_x + m c^2 σ_z

def build_H_dirac(N=200, x_min=-5.0, x_max=5.0, m=1.0):

    x = np.linspace(x_min, x_max, N)

    dx = x[1] - x[0]

    D = (np.eye(N, k=1) - np.eye(N, k=-1)) / (2*dx)

    p_op = -1j * hbar * D

# Version pédagogique (2D)

 

import sympy as sp

 

# -----------------------------

# 1. Coordonnées

# -----------------------------

t, x = sp.symbols('t x')

 

# -----------------------------

# 2. Métrique générique 2D

#    g = diag(-A(t,x), B(t,x))

# -----------------------------

A = sp.Function('A')(t, x)

B = sp.Function('B')(t, x)

 

g = sp.Matrix([[-A, 0],

               [ 0, B]])

 

g_inv = g.inv()

 

# -----------------------------

# 3. Calcul du scalaire de courbure R

# -----------------------------

# Christoffel symbols

Gamma = [[[0 for k in range(2)] for j in range(2)] for i in range(2)]

 

coords = [t, x]

 

for i in range(2):

    for j in range(2):

        for k in range(2):

            Gamma[i][j][k] = sp.simplify(

                0.5 * sum(

                    g_inv[i, l] * (

                        sp.diff(g[l, j], coords[k]) +

                        sp.diff(g[l, k], coords[j]) -

                        sp.diff(g[j, k], coords[l])

                    )

                    for l in range(2)

                )

            )

 

# Riemann tensor (contracted)

R = 0

for i in range(2):

    for j in range(2):

        R += g_inv[i, j] * (

            sp.diff(Gamma[k][i][j], coords[k]) -

            sp.diff(Gamma[k][i][k], coords[j])

        )

 

R = sp.simplify(R)

 

print("Scalaire de courbure R(t,x) =")

sp.pprint(R)

 

# -----------------------------

# 4. Résistance géométrique

# -----------------------------

R_geo = sp.simplify(sp.sqrt(A*B) * R)

print("\nRésistance géométrique locale = sqrt(det(g)) * R =")

sp.pprint(R_geo)
